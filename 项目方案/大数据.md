```plain
 public LocationApiVo getLngAndLatByAddress(String address){
 String url = "https://api.map.baidu.com/geocoding/v3/?address="+address+"&output=json&ak="+ak+"&callback=showLocation";

 log.info("getLngAndLatByAddress url====>{}",url);
 try {
 ResponseEntity<String> exchange = restTemplate.exchange(url, HttpMethod.GET, null, String.class);
 if (HttpStatus.HTTP_OK != exchange.getStatusCodeValue()){
 log.error("getLngAndLatByAddress: {}", exchange.getBody());
 throw new OrderBusinessException(OrderBusinessExceptionEnum.INTERFACE_SYSTEM_ERROR);
 }
 String body = exchange.getBody();
 String s = StrUtil.subBetween(body, "(", ")");
 JSONObject jsonObject = JSONObject.parseObject(s);
 int status = jsonObject.getIntValue("status");
 if (0 != status){
 log.error("getLngAndLatByAddress: {}", exchange.getBody());
 return null;
// throw new OrderBusinessException(OrderBusinessExceptionEnum.INTERFACE_SYSTEM_ERROR);
 }
 String result = jsonObject.getString("result");
 LocationApiVo location = JSONObject.parseObject(JSONObject.parseObject(result).getString("location"), LocationApiVo.class);
 location.setAddress(address);
 return location;
 }catch (Exception e) {
 log.error(e.getMessage(), e);
 throw new OrderBusinessException(OrderBusinessExceptionEnum.INTERFACE_SYSTEM_ERROR);
 }

 }
```
