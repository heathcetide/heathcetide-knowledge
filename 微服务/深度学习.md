```plain
java

import com.huaban.analysis.jieba.JiebaSegmenter;
import org.apache.commons.lang3.StringUtils;
import org.deeplearning4j.datasets.iterator.impl.ListDataSetIterator;
import org.nd4j.linalg.dataset.api.iterator.DataSetIterator;
import org.nd4j.linalg.dataset.api.preprocessor.NormalizerStandardize;

import java.io.*;
import java.nio.charset.StandardCharsets;
import java.util.*;

public class DataPreprocessor {
    private static JiebaSegmenter segmenter = new JiebaSegmenter();
    private static Set<String> stopWords = loadStopWords("src/main/resources/stopwords.txt");

    public static List<Sample> loadData(String filePath) throws IOException {
        List<Sample> samples = new ArrayList<>();
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(new FileInputStream(filePath), StandardCharsets.UTF_8))) {
            String line;
            while ((line = reader.readLine()) != null) {
                String[] parts = line.split("\t");
                if (parts.length != 2) continue;
                String label = parts[0];
                String text = preprocess(parts[1]);
                samples.add(new Sample(label, text));
            }
        }
        return samples;
    }

    private static String preprocess(String text) {
        // 分词
        List<String> segments = segmenter.sentenceProcess(text);
        // 转小写
        List<String> lowerSegments = new ArrayList<>();
        for (String seg : segments) {
            lowerSegments.add(seg.toLowerCase());
        }
        // 去除停用词和非中文字符
        List<String> filtered = new ArrayList<>();
        for (String word : lowerSegments) {
            if (!stopWords.contains(word) && word.matches("[\\u4e00-\\u9fa5a-zA-Z0-9]+")) {
                filtered.add(word);
            }
        }
        return String.join(" ", filtered);
    }

    private static Set<String> loadStopWords(String filePath) {
        Set<String> stopWords = new HashSet<>();
        try (BufferedReader reader = new BufferedReader(new InputStreamReader(new FileInputStream(filePath), StandardCharsets.UTF_8))) {
            String word;
            while ((word = reader.readLine()) != null) {
                stopWords.add(word.trim());
            }
        } catch (IOException e) {
            System.err.println("无法加载停用词文件: " + filePath);
        }
        return stopWords;
    }

    // 内部类表示一个样本
    public static class Sample {
        private String label;
        private String text;

        public Sample(String label, String text) {
            this.label = label;
            this.text = text;
        }

        public String getLabel() {
            return label;
        }

        public String getText() {
            return text;
        }
    }
}
```

## 3. 词向量（Word Embeddings）
词向量将文本中的词语映射到高维空间，捕捉词语的语义信息。我们可以使用 DL4J 提供的 `Word2Vec` 实现，或加载预训练的词向量。

### 3.1. 使用 DL4J 训练 Word2Vec
```plain
java


复制代码
import org.deeplearning4j.models.embeddings.loader.WordVectorSerializer;
import org.deeplearning4j.models.embeddings.wordvectors.WordVectors;
import org.deeplearning4j.models.word2vec.Word2Vec;
import org.deeplearning4j.text.sentenceiterator.CollectionSentenceIterator;
import org.deeplearning4j.text.tokenization.tokenizerfactory.DefaultTokenizerFactory;

import java.util.List;

public class WordEmbeddings {
    public static Word2Vec trainWord2Vec(List<DataPreprocessor.Sample> samples) {
        List<String> sentences = new ArrayList<>();
        for (DataPreprocessor.Sample sample : samples) {
            sentences.add(sample.getText());
        }

        CollectionSentenceIterator iterator = new CollectionSentenceIterator(sentences);
        Word2Vec vec = new Word2Vec.Builder()
                .minWordFrequency(5)
                .iterations(1)
                .layerSize(100)
                .seed(42)
                .windowSize(5)
                .iterate(iterator)
                .tokenizerFactory(new DefaultTokenizerFactory())
                .build();
        vec.fit();
        return vec;
    }

    public static WordVectors loadPretrainedWord2Vec(String filePath) throws IOException {
        return WordVectorSerializer.readWord2VecModel(new File(filePath));
    }
}
```

## 4. 构建 CNN 模型
使用 DL4J 构建一个简单的 CNN 模型用于文本分类。下面的示例展示了如何定义模型架构。

```plain
java


复制代码
import org.deeplearning4j.nn.api.OptimizationAlgorithm;
import org.deeplearning4j.nn.conf.*;
import org.deeplearning4j.nn.conf.layers.*;
import org.deeplearning4j.nn.multilayer.MultiLayerNetwork;
import org.nd4j.linalg.activations.Activation;
import org.nd4j.linalg.learning.config.Adam;

public class CNNModelBuilder {
    public static MultiLayerNetwork buildModel(int vocabSize, int embeddingSize, int outputClasses, int maxSentenceLength) {
        int[] kernelSizes = new int[]{3, 4, 5}; // 多种卷积核尺寸
        int numFilters = 100; // 每种卷积核的过滤器数量

        ComputationGraphConfiguration.GraphBuilder graphBuilder = new NeuralNetConfiguration.Builder()
                .seed(123)
                .optimizationAlgo(OptimizationAlgorithm.STOCHASTIC_GRADIENT_DESCENT)
                .updater(new Adam(1e-3))
                .graphBuilder()
                .addInputs("input")
                .setInputTypes(InputType.convolutionalFlat(maxSentenceLength, embeddingSize, 1));

        // 卷积层和池化层
        for (int i = 0; i < kernelSizes.length; i++) {
            graphBuilder.addLayer("cnn" + i,
                    new ConvolutionLayer.Builder()
                            .kernelSize(kernelSizes[i], embeddingSize)
                            .stride(1, 1)
                            .nOut(numFilters)
                            .activation(Activation.RELU)
                            .build(),
                    "input");
            graphBuilder.addLayer("globalPool" + i,
                    new GlobalPoolingLayer.Builder(PoolingType.MAX).build(),
                    "cnn" + i);
        }

        // 合并池化层输出
        String[] globalPoolLayerNames = Arrays.stream(kernelSizes)
                .mapToObj(i -> "globalPool" + (Arrays.asList(kernelSizes).indexOf(i)))
                .toArray(String[]::new);
        graphBuilder.addVertex("concat", new MergeVertex(), globalPoolLayerNames);

        // 全连接层
        graphBuilder.addLayer("dense",
                new DenseLayer.Builder()
                        .activation(Activation.RELU)
                        .nOut(128)
                        .build(),
                "concat");

        // 输出层
        graphBuilder.addLayer("output",
                new OutputLayer.Builder(LossFunctions.LossFunction.NEGATIVELOGLIKELIHOOD)
                        .activation(Activation.SOFTMAX)
                        .nOut(outputClasses)
                        .build(),
                "dense");

        // 设置輸出
        graphBuilder.setOutputs("output");

        ComputationGraphConfiguration config = graphBuilder.build();

        MultiLayerNetwork model = new MultiLayerNetwork(config);
        model.init();
        return model;
    }
}
```

**说明**：

+ **输入层**：将文本表示为二维矩阵，维度为 `(maxSentenceLength, embeddingSize)`。
+ **卷积层**：使用多种卷积核尺寸（例如 3、4、5），每种卷积核提取不同 n-gram 的特征。
+ **池化层**：对每个卷积层的输出进行全局最大池化。
+ **合并层**：将所有池化层的输出合并。
+ **全连接层**：添加一个全连接层提取高层特征。
+ **输出层**：使用 Softmax 激活函数进行多类别分类。

## 5. 训练模型
### 5.1. 将文本转换为向量
我们需要将每个句子转换为固定长度的向量，通常通过填充或截断实现。以下示例展示了如何进行这一转换。

```plain
java


复制代码
import org.deeplearning4j.datasets.iterator.DataSetIterator;
import org.deeplearning4j.nn.api.Model;
import org.deeplearning4j.nn.graph.ComputationGraph;
import org.nd4j.linalg.api.ndarray.INDArray;
import org.nd4j.linalg.dataset.api.DataSet;
import org.nd4j.linalg.dataset.api.iterator.BaseDatasetIterator;
import org.nd4j.linalg.dataset.api.preprocessor.DataNormalization;
import org.nd4j.linalg.factory.Nd4j;

import java.util.List;

public class DataVectorizer {
    private static final int maxSentenceLength = 100; // 最大句子长度
    private Word2Vec word2Vec;

    public DataVectorizer(Word2Vec word2Vec) {
        this.word2Vec = word2Vec;
    }

    public DataSetIterator createIterator(List<DataPreprocessor.Sample> samples, int batchSize, Map<String, Integer> labelMap) {
        List<DataSet> dataSets = new ArrayList<>();
        for (DataPreprocessor.Sample sample : samples) {
            INDArray features = vectorizeText(sample.getText());
            INDArray labels = Nd4j.zeros(labelMap.size());
            labels.putScalar(labelMap.get(sample.getLabel()), 1.0);
            dataSets.add(new DataSet(features, labels));
        }
        return new ListDataSetIterator<>(dataSets, batchSize);
    }

    private INDArray vectorizeText(String text) {
        String[] words = text.split(" ");
        INDArray vector = Nd4j.zeros(maxSentenceLength, word2Vec.getLayerSize());
        for (int i = 0; i < Math.min(words.length, maxSentenceLength); i++) {
            if (word2Vec.hasWord(words[i])) {
                vector.putRow(i, word2Vec.getWordVectorMatrix(words[i]));
            }
        }
        return vector.reshape(1, maxSentenceLength * word2Vec.getLayerSize());
    }
}
```

### 5.2. 模型训练

```plain
import org.deeplearning4j.nn.graph.ComputationGraph;

import org.nd4j.linalg.dataset.api.iterator.DataSetIterator;

        }
    }
}
```

## 6. 评估模型
使用测试数据评估模型的性能。

```plain
java


复制代码
import org.deeplearning4j.eval.Evaluation;
import org.deeplearning4j.nn.graph.ComputationGraph;
import org.nd4j.linalg.dataset.api.iterator.DataSetIterator;

import java.util.List;

public class Evaluator {
    public static void evaluateModel(ComputationGraph model, DataSetIterator testIter, List<String> labelNames) {
        Evaluation eval = new Evaluation(labelNames);
        while (testIter.hasNext()) {
            DataSet batch = testIter.next();
            INDArray output = model.outputSingle(batch.getFeatures());
            eval.eval(batch.getLabels(), output);
        }
        System.out.println(eval.stats());
    }
}
```

```plain
import org.deeplearning4j.nn.graph.ComputationGraph;
import org.deeplearning4j.models.word2vec.Word2Vec;
import org.nd4j.linalg.dataset.api.iterator.DataSetIterator;

import java.io.IOException;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

public class SentimentModelTrainer {
    public static void main(String[] args) {
        String trainingFilePath = "src/test/sentiment-training.txt"; // 训练数据文件路径
        String modelOutputPath = "sentiment-cnn-model.zip";        // 模型保存路径

        try {
            // 步骤 1: 数据预处理
            List<DataPreprocessor.Sample> samples = DataPreprocessor.loadData(trainingFilePath);
            System.out.println("加载和预处理数据完成。样本数量: " + samples.size());

            // 步骤 2: 训练词向量
            Word2Vec word2Vec = WordEmbeddings.trainWord2Vec(samples);
            System.out.println("词向量训练完成。");

            // 步骤 3: 准备标签映射
            Map<String, Integer> labelMap = new HashMap<>();
            int labelIndex = 0;
            for (DataPreprocessor.Sample sample : samples) {
                if (!labelMap.containsKey(sample.getLabel())) {
                    labelMap.put(sample.getLabel(), labelIndex++);
                }
            }
            List<String> labelNames = new ArrayList<>(labelMap.keySet());
            System.out.println("标签映射: " + labelMap);

            // 步骤 4: 向量化数据
            DataVectorizer vectorizer = new DataVectorizer(word2Vec);
            DataSetIterator trainIter = vectorizer.createIterator(samples, 32, labelMap);
            System.out.println("数据向量化完成。");

            // 步骤 5: 构建模型
            int vocabSize = word2Vec.getVocab().size();
            int embeddingSize = word2Vec.getLayerSize();
            int outputClasses = labelMap.size();
            int maxSentenceLength = 100; // 与 DataVectorizer 中的一致

            ComputationGraph model = CNNModelBuilder.buildModel(vocabSize, embeddingSize, outputClasses, maxSentenceLength);
            System.out.println("CNN 模型构建完成。");

            // 步骤 6: 训练模型
            Trainer.trainModel(model, trainIter, 10); // 训练10个Epoch
            System.out.println("模型训练完成。");

            // 步骤 7: 评估模型（这里假设使用相同的数据作为测试）
            Evaluator.evaluateModel(model, trainIter, labelNames);

            // 步骤 8: 保存模型
            ModelHandler.saveModel(model, modelOutputPath);
            System.out.println("模型已保存到: " + modelOutputPath);

        } catch (IOException e) {
            e.printStackTrace();
        }
    }
}
```

**注意**：

+ **数据集划分**：在实际应用中，您应该将数据集划分为训练集和测试集，以评估模型的泛化能力。上述示例中简化为使用相同的数据进行训练和评估，仅供参考。
+ **批次大小和 Epoch 数量**：根据您的数据量和计算资源调整批次大小和训练轮数（Epoch）。
+ **词向量训练时间**：词向量的训练可能需要一些时间，尤其是在大型数据集上。您可以选择加载预训练的词向量以节省时间。
